import gradio as gr def saludar(nombre): if not nombre.strip(): return "Por favor, ingresá tu nombre." return f"¡Hola, {nombre}!" with gr.Blocks(title="Mi aplicación") as demo: gr.Markdown("# Mi aplicación con Gradio Lite") gr.Markdown("Esta aplicación se ejecuta directamente en el navegador.") nombre = gr.Textbox( label="Nombre", placeholder="Escribí tu nombre..." ) boton = gr.Button("Saludar", variant="primary") resultado = gr.Textbox(label="Resultado") boton.click( fn=saludar, inputs=nombre, outputs=resultado ) demo.launch()
